public class DatabaseManager {
    private static final String DATA_FILE = "repair_data.csv";
    private static final DateTimeFormatter displayFormatter = DateTimeFormatter.ofPattern("yyyy/MM/dd");
    private List<RepairRecord> records;
    private static DatabaseManager instance;
    
    private DatabaseManager() {
        records = new ArrayList<>();
        loadData();
    }
    
    public static synchronized DatabaseManager getInstance() {
        if (instance == null) {
            instance = new DatabaseManager();

    public void addRecord(RepairRecord record) {
        records.add(record);
        saveData();
    }
    
    public void updateRecord(RepairRecord record) {
        for (int i = 0; i < records.size(); i++) {
            if (records.get(i).getId() == record.getId()) {
                records.set(i, record);
                saveData();
                return;
            }
        }
    }
    
    public void deleteRecord(int id) {
        records.removeIf(r -> r.getId() == id);
        saveData();
    }
    
    public List<RepairRecord> getAllRecords() {

    public List<RepairRecord> getFilteredRecords(String dateFilter, String machineFilter, 
                                                  String spindleFilter, String errorFilter) {
        List<RepairRecord> filtered = new ArrayList<>();
        for (RepairRecord r : records) {
            boolean match = true;
            if (dateFilter != null && !dateFilter.isEmpty() && !dateFilter.equals("All Dates")) {
                if (r.getBreakDownDate() == null || !r.getBreakDownDateString().equals(dateFilter)) match = false;
            }
            if (machineFilter != null && !machineFilter.isEmpty() && !machineFilter.equals("All")) {
                if (!machineFilter.equals(r.getMachineNo())) match = false;
            }
            if (spindleFilter != null && !spindleFilter.isEmpty() && !spindleFilter.equals("All")) {
                if (r.getSpindle() == null || !r.getSpindle().equals(spindleFilter)) match = false;
            }
            if (errorFilter != null && !errorFilter.isEmpty() && !errorFilter.equals("All Errors")) {
                if (r.getRepairError() == null || !r.getRepairError().equals(errorFilter)) match = false;
            }
            if (match) filtered.add(r);
        }
        return filtered;
    }
    
    public List<String> getUniqueMachines() {
        Set<String> machines = new TreeSet<>();
        for (RepairRecord r : records) {
            if (r.getMachineNo() != null && !r.getMachineNo().isEmpty() && 
                !r.getMachineNo().contains(",") && !r.getMachineNo().toLowerCase().contains("machine")) {
                machines.add(r.getMachineNo());
            }
        }
        return new ArrayList<>(machines);
    }
    
    public List<String> getUniqueSpindles() {
        Set<String> spindles = new TreeSet<>(new Comparator<String>() {
            @Override
            public int compare(String s1, String s2) {
                try {
                    int i1 = Integer.parseInt(s1);
                    int i2 = Integer.parseInt(s2);
                    return Integer.compare(i1, i2);
                } catch (NumberFormatException e) {
                    return s1.compareTo(s2);
                }
            }
        });
        for (RepairRecord r : records) {
            if (r.getSpindle() != null && !r.getSpindle().isEmpty() && 
                !r.getSpindle().contains(",") && !r.getSpindle().toLowerCase().contains("spindle") &&
                !r.getSpindle().contains("සිට")) {
                try {
                    Integer.parseInt(r.getSpindle());
                    spindles.add(r.getSpindle());
                } catch (NumberFormatException e) {}
            }
        }
        return new ArrayList<>(spindles);
    }
    
    public List<String> getUniqueSamples() {

    public double calculateDailyPercentage(LocalDate date) {
        List<RepairRecord> dayRecords = new ArrayList<>();
        for (RepairRecord r : records) {
            if (r.getBreakDownDate() != null && r.getBreakDownDate().equals(date)) {
                dayRecords.add(r);
            }
        }
        if (dayRecords.isEmpty()) return 0.0;
        
        long errorCount = dayRecords.stream()
            .filter(r -> r.getRepairError() != null && !r.getRepairError().isEmpty() 
                && !r.getRepairError().equalsIgnoreCase("No Error")
                && !r.getRepairError().equalsIgnoreCase("no Error"))
            .count();
        return (errorCount * 100.0) / dayRecords.size();
    }
    
    public Map<String, Double> calculateWeeklyPercentage(LocalDate weekStart) {
        Map<String, Double> result = new LinkedHashMap<>();
        String[] days = {"Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"};
        for (int i = 0; i < 7; i++) {
            LocalDate day = weekStart.plusDays(i);
            double pct = calculateDailyPercentage(day);
            result.put(day.format(displayFormatter) + " (" + days[i] + ")", pct);
        }
        return result;
    }

    private void loadData() {
        File file = new File(DATA_FILE);
        if (!file.exists()) {
            System.out.println("Data file not found: " + DATA_FILE);
            return;
        }
        
        try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
            String line = reader.readLine();
            int maxId = 0;
            
            while ((line = reader.readLine()) != null) {
                if (line.trim().isEmpty()) continue;
                
                String[] parts = parseCSVLine(line);
                if (parts.length < 5) continue;
                
                try {
                    RepairRecord r = new RepairRecord();
                    r.setId(++maxId);
                    
                    if (parts.length > 0 && !parts[0].trim().isEmpty()) {
                        r.setBreakDownDate(RepairRecord.parseDate(parts[0].trim()));
                    }
                    if (parts.length > 1) {
                        r.setMachineNo(parts[1].trim());
                    }
                    if (parts.length > 2) {
                        r.setSpindle(parts[2].trim());
                    }
                    if (parts.length > 3) {
                        r.setRepairStatus(parts[3].trim());
                    }
                    if (parts.length > 4) {
                        r.setAirCoveringError(parts[4].trim());
                    }
                    if (parts.length > 5) {
                        r.setRepairError(parts[5].trim());
                    }
                    if (parts.length > 6) {
                        r.setRepairReason(parts[6].trim());
                    }
                    if (parts.length > 7 && !parts[7].trim().isEmpty()) {
                        r.setRepairDate(RepairRecord.parseDate(parts[7].trim()));
                    }
                    if (parts.length > 8) {
                        r.setSampleNo(parts[8].trim());
                    }
                    
                    records.add(r);
                    
                } catch (Exception e) {
                    System.err.println("Error parsing line: " + line + " - " + e.getMessage());
                }
            }
            
            RepairRecord.resetIdCounter();
            for (int i = 0; i < maxId; i++) {
                RepairRecord rec = new RepairRecord();
                rec.setId(0);
            }
            
            System.out.println("Loaded " + records.size() + " records from " + DATA_FILE);
            
        } catch (IOException e) {
            System.err.println("Error loading data: " + e.getMessage());
        }
    }
    
    private void saveData() {
        try (PrintWriter writer = new PrintWriter(new FileWriter(DATA_FILE))) {
            writer.println("Break Down Date,Machine no,Spiudle,R.E,Air Covering Error,Repair Error,Repair Reason,Repair Date,Sampal no");
            for (RepairRecord r : records) {
                writer.printf("%s,%s,%s,%s,%s,%s,%s,%s,%s%n",
                    r.getBreakDownDateString(),
                    escapeCSV(r.getMachineNo()),
                    escapeCSV(r.getSpindle()),
                    escapeCSV(r.getRepairStatus()),
                    escapeCSV(r.getAirCoveringError()),
                    escapeCSV(r.getRepairError()),
                    escapeCSV(r.getRepairReason()),
                    r.getRepairDateString(),
                    escapeCSV(r.getSampleNo())
                );
            }
        } catch (IOException e) {
            System.err.println("Error saving data: " + e.getMessage());
        }
    }
    
    private String escapeCSV(String value) {
        if (value == null) return "";
        if (value.contains(",") || value.contains("\"") || value.contains("\n")) {
            return "\"" + value.replace("\"", "\"\"") + "\"";
        }
        return value;
    }
    
    private String[] parseCSVLine(String line) {
        List<String> tokens = new ArrayList<>();
        StringBuilder sb = new StringBuilder();
        boolean inQuotes = false;
        
        for (int i = 0; i < line.length(); i++) {
            char c = line.charAt(i);
            if (c == '"') {
                if (inQuotes && i + 1 < line.length() && line.charAt(i + 1) == '"') {
                    sb.append('"');
                    i++;
                } else {
                    inQuotes = !inQuotes;
                }
            } else if (c == ',' && !inQuotes) {
                tokens.add(sb.toString().trim());
                sb = new StringBuilder();
            } else {
                sb.append(c);
            }
        }
        tokens.add(sb.toString().trim());
        
        return tokens.toArray(new String[0]);
    }
    
    public void clearAll() {
        records.clear();
        RepairRecord.resetIdCounter();
        saveData();
    }
}

        int id = (Integer) tableModel.getValueAt(selectedRow, 0);
        int confirm = JOptionPane.showConfirmDialog(this,
            "Are you sure you want to delete record ID: " + id + "?",
            "Confirm Delete", JOptionPane.YES_NO_OPTION, JOptionPane.WARNING_MESSAGE);
        
        if (confirm == JOptionPane.YES_OPTION) {
            DatabaseManager.getInstance().deleteRecord(id);
            refreshTable();
            JOptionPane.showMessageDialog(this, "Record deleted successfully!");
        }
    }
    
    private void updatePercentage() {
        String selectedDate = filterDateCombo.getSelectedIndex() > 0 ? 
            (String) filterDateCombo.getSelectedItem() : null;
        
        if (selectedDate != null) {
            try {
                LocalDate date = LocalDate.parse(selectedDate, formatter);
                double pct = DatabaseManager.getInstance().calculateDailyPercentage(date);
                int errorCount = DatabaseManager.getInstance().getErrorCountForDate(date);
                int totalCount = DatabaseManager.getInstance().getTotalCountForDate(date);
                percentageLabel.setText(String.format("📈 Daily Error %%: %.2f%% (%d/%d)", pct, errorCount, totalCount));
            } catch (Exception e) {
                percentageLabel.setText("📈 Daily Error %: N/A");
            }
        } else {
            percentageLabel.setText("📈 Daily Error %: Select a date");
        }
    }

    private void populateWeekCombo() {
        LocalDate today = LocalDate.now();
        LocalDate startOfPeriod = LocalDate.of(2026, 7, 1);
        LocalDate endDate = today.plusWeeks(2);
        
        LocalDate currentWeek = startOfPeriod.with(TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY));
        
        while (!currentWeek.isAfter(endDate)) {
            LocalDate weekEnd = currentWeek.plusDays(6);
            String weekLabel = String.format("Week of %s - %s", 
                currentWeek.format(formatter), weekEnd.format(formatter));
            weekCombo.addItem(weekLabel);
            currentWeek = currentWeek.plusWeeks(1);
        }
    }
